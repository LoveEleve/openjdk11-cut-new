# GDB 脚本 - 验证 Universe::create_heap() 调用链
# 使用方法: gdb -x gdb_create_heap.txt --args java -Xms8g -Xmx8g -XX:+UseG1GC -version

set pagination off
set print pretty on

# 使用 pending breakpoints（等待共享库加载）
set breakpoint pending on

# 在关键函数入口打断点
b Universe::create_heap
b GCConfig::arguments
b G1Arguments::create_heap
b G1CollectedHeap::G1CollectedHeap
b G1CollectorPolicy::G1CollectorPolicy

# 运行
run

printf "\n========== 1. Universe::create_heap() 入口 ==========\n"
bt 3

# 继续到 GCConfig::arguments
c

printf "\n========== 2. GCConfig::arguments() 多态分发 ==========\n"
# 打印实际返回的对象类型
p GCConfig::arguments()
p *GCConfig::arguments()

# 继续到 G1Arguments::create_heap
c

printf "\n========== 3. G1Arguments::create_heap() ==========\n"
bt 3

# 继续到 G1CollectorPolicy 构造函数
c

printf "\n========== 4. G1CollectorPolicy 构造函数 ==========\n"
# 打印 this 指针和大小
p this
p sizeof(G1CollectorPolicy)

# 继续到 G1CollectedHeap 构造函数
c

printf "\n========== 5. G1CollectedHeap 构造函数 ==========\n"
# 打印 this 指针和大小
p this
p sizeof(G1CollectedHeap)

printf "\n========== 6. 关键数据结构大小汇总 ==========\n"
p sizeof(CollectedHeap)
p sizeof(G1CollectedHeap)
p sizeof(G1CollectorPolicy)
p sizeof(HeapRegion)
p sizeof(G1RemSet)
p sizeof(G1ConcurrentMark)

printf "\n========== 7. G1CollectedHeap 关键字段偏移 ==========\n"
p &((G1CollectedHeap*)0)->_g1_policy
p &((G1CollectedHeap*)0)->_hrm
p &((G1CollectedHeap*)0)->_g1_remset
p &((G1CollectedHeap*)0)->_cm
p &((G1CollectedHeap*)0)->_cr

printf "\n========== 验证完成 ==========\n"
quit
