# 在 compilation_init_phase2 后断点（此时编译系统已完全初始化）

    printf "\n--- C1 Compiler ---\n"
    printf "name: %s\n", ((AbstractCompiler*)CompileBroker::_compilers[0])->name()

    printf "name: %s\n", ((AbstractCompiler*)CompileBroker::_compilers[1])->name()

printf "_c1_compile_queue: %p\n", CompileBroker::_c1_compile_queue
printf "_c2_compile_queue: %p\n", CompileBroker::_c2_compile_queue

    printf "\n--- C1 Compile Queue ---\n"
    printf "name: %s\n", ((CompileQueue*)CompileBroker::_c1_compile_queue)->_name
    printf "size: %d\n", ((CompileQueue*)CompileBroker::_c1_compile_queue)->_size

    printf "\n--- C2 Compile Queue ---\n"
    printf "name: %s\n", ((CompileQueue*)CompileBroker::_c2_compile_queue)->_name
    printf "size: %d\n", ((CompileQueue*)CompileBroker::_c2_compile_queue)->_size
    printf "first: %p\n", ((CompileQueue*)CompileBroker::_c2_compile_queue)->_first
end

printf "\n========== 5. 编译器线程对象数组 ==========\n"
printf "_compiler1_objects: %p\n", CompileBroker::_compiler1_objects
printf "_compiler2_objects: %p\n", CompileBroker::_compiler2_objects

printf "\n========== 6. 编译 ID 计数器 ==========\n"
printf "_compilation_id: %d\n", CompileBroker::_compilation_id

printf "_perf_total_compile_count: %p\n", CompileBroker::_perf_total_compile_count
printf "_perf_total_bailout_count: %p\n", CompileBroker::_perf_total_bailout_count

printf "_total_compile_count: %d\n", CompileBroker::_total_compile_count
printf "_total_bailout_count: %d\n", CompileBroker::_total_bailout_count
printf "_total_invalidated_count: %d\n", CompileBroker::_total_invalidated_count
printf "_total_osr_compile_count: %d\n", CompileBroker::_total_osr_compile_count
