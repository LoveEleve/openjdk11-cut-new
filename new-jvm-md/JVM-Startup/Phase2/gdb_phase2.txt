# Phase 2 GDB 调试脚本 - os::init_2() 和 vm_init_globals()
# 使用方法:
# gdb -x jvm-md/Phase2/gdb_phase2.txt ./build/linux-x86_64-normal-server-slowdebug/jdk/bin/java

set pagination off
set print pretty on

#################################################
# Part 1: os::init_2() 调试
#################################################

printf "\n========== Phase 2.1: os::init_2() ==========\n\n"

# 在 os::init_2 入口设置断点
b os::init_2
run -Xms256m -Xmx256m -XX:+UseG1GC -Xint -version

printf "\n>>> Entering os::init_2()\n"
printf ">>> Current thread: main Java launcher thread\n"

# 查看 SR_signum
printf "\n========== SR Signal Config ==========\n"
# 注意: SR_signum 是静态变量，需要在 SR_initialize 之后查看
b SR_initialize
c
finish

printf "\n>>> SR_initialize() completed\n"
printf "SR_signum (suspend/resume signal): %d (should be SIGUSR2=12)\n", SR_signum

# 查看信号集初始化
b os::Linux::signal_sets_init
c
finish

printf "\n>>> signal_sets_init() completed\n"
printf ">>> unblocked_sigs and vm_sigs are now configured\n"

# 继续到 install_signal_handlers
b os::Linux::install_signal_handlers
c
finish

printf "\n>>> install_signal_handlers() completed\n"
printf ">>> SIGSEGV, SIGFPE, SIGPIPE, etc. handlers installed\n"

# os::init_2 完成
c

printf "\n>>> os::init_2() completed successfully\n"

#################################################
# Part 2: vm_init_globals() 调试
#################################################

printf "\n========== Phase 2.2: vm_init_globals() ==========\n\n"

# 在 vm_init_globals 入口设置断点
b vm_init_globals
c

printf "\n>>> Entering vm_init_globals()\n"

# 单步进入 basic_types_init
b basic_types_init
c
finish

# 查看指针压缩相关变量
printf "\n========== Basic Types (after basic_types_init) ==========\n"
printf "heapOopSize: %d bytes\n", heapOopSize
printf "BytesPerHeapOop: %d\n", BytesPerHeapOop
printf "LogBytesPerHeapOop: %d\n", LogBytesPerHeapOop
printf "BitsPerHeapOop: %d\n", BitsPerHeapOop

# 进入 mutex_init
b mutex_init
c
finish

# 查看部分全局锁
printf "\n========== Global Mutexes (after mutex_init) ==========\n"
printf "Heap_lock: %p\n", Heap_lock
printf "Threads_lock: %p\n", Threads_lock
printf "CodeCache_lock: %p\n", CodeCache_lock
printf "Safepoint_lock: %p\n", Safepoint_lock
printf "VMOperationQueue_lock: %p\n", VMOperationQueue_lock
printf "SystemDictionary_lock: %p\n", SystemDictionary_lock
printf "Total mutexes: %d\n", _num_mutex

# 进入 chunkpool_init
b chunkpool_init
c
finish

printf "\n========== ChunkPool (after chunkpool_init) ==========\n"
printf "ChunkPool::_tiny_pool: %p\n", ChunkPool::_tiny_pool
printf "ChunkPool::_small_pool: %p\n", ChunkPool::_small_pool
printf "ChunkPool::_medium_pool: %p\n", ChunkPool::_medium_pool
printf "ChunkPool::_large_pool: %p\n", ChunkPool::_large_pool

# 进入 perfMemory_init
b perfMemory_init
c
finish

# 查看 PerfMemory 状态
printf "\n========== PerfMemory (after perfMemory_init) ==========\n"
printf "PerfMemory::_start: %p\n", PerfMemory::_start
printf "PerfMemory::_end: %p\n", PerfMemory::_end
printf "PerfMemory::_top: %p\n", PerfMemory::_top
printf "PerfMemory::_capacity: %lu bytes\n", PerfMemory::_capacity
printf "PerfMemory::_initialized: %d\n", PerfMemory::_initialized

# 查看 PerfDataPrologue 内容
if PerfMemory::_prologue != 0
    printf "\n========== PerfDataPrologue ==========\n"
    printf "magic: 0x%x\n", PerfMemory::_prologue->magic
    printf "major_version: %d\n", PerfMemory::_prologue->major_version
    printf "minor_version: %d\n", PerfMemory::_prologue->minor_version
    printf "accessible: %d\n", PerfMemory::_prologue->accessible
    printf "entry_offset: %d\n", PerfMemory::_prologue->entry_offset
    printf "num_entries: %d\n", PerfMemory::_prologue->num_entries
end

# 完成
printf "\n========== Phase 2 Completed ==========\n"
printf "os::init_2() - Signal handlers installed\n"
printf "vm_init_globals() - Basic types, mutexes, pools initialized\n"

quit
