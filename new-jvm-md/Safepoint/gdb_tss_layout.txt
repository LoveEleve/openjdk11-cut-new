# ========== ThreadSafepointState 字段偏移 ==========

printf "_has_called_back: %d\n", $tss->_has_called_back

printf "_type: %d (0=_running, 1=_at_safepoint, 2=_call_back)\n", $tss->_type
printf "_orig_thread_state: %d\n", $tss->_orig_thread_state

printf "\n========== JavaThread polling page ==========\n"
