### 1.2 核心锁机制分析

```c
// jsig.c:98-111

static void signal_lock() {
  pthread_mutex_lock(&mutex);
  /* When the jvm is installing its set of signal handlers, threads
   * other than the jvm thread should wait. */
  if (jvm_signal_installing) {
    if (tid != pthread_self()) {
      pthread_cond_wait(&cond, &mutex);
    }
  }
}

static void signal_unlock() {
  pthread_mutex_unlock(&mutex);
}
```

**逐行解析**：

```c
pthread_mutex_lock(&mutex);   // [第1行] 获取互斥锁，保证串行访问

if (jvm_signal_installing) {  // [第2行] 检查 JVM 是否正在批量安装信号
    if (tid != pthread_self()) {  // [第3行] 如果不是 JVM 线程自身
        pthread_cond_wait(&cond, &mutex);  // [第4行] 释放锁并等待条件变量
    }
}
```

**为什么需要这个机制？**

```
场景：JVM 启动时需要安装多个信号处理器（SIGSEGV, SIGBUS, SIGILL...）

无此机制时：
  线程A: sigaction(SIGSEGV, ...) → 保存旧处理器，安装新处理器
  线程B: sigaction(SIGSEGV, ...) → 在A完成前插入，保存了A的新处理器作为"旧处理器"
  结果：信号链断裂，JVM 处理器丢失

有此机制时：
  JVM: JVM_begin_signal_setting() → jvm_signal_installing = true
  JVM: 安装 SIGSEGV, SIGBUS...（其他线程被阻塞）
  JVM: JVM_end_signal_setting() → jvm_signal_installing = false, 广播条件变量
  其他线程：被唤醒，可以正常执行
```

### 1.4 核心：sigaction() 拦截函数

```c
// jsig.c:248-302

int sigaction(int sig, const struct sigaction *act, struct sigaction *oact) {
  int res;
  bool sigused;
  struct sigaction oldAct;

  if (sig < 0 || sig >= MAX_SIGNALS) {  // [1] 参数检查
    errno = EINVAL;
    return -1;
  }

#ifdef MACOSX
  if (reentry) {                        // [2] macOS 重入保护
    return call_os_sigaction(sig, act, oact);
  }
#endif

  signal_lock();                        // [3] 获取锁，可能等待

  allocate_sact();                      // [4] 延迟分配 sact 数组（Solaris）
  sigused = sigismember(&jvmsigs, sig); // [5] 检查信号是否被 JVM 占用
  
  // ========== 分支 1: JVM 已安装且占用此信号 ==========
  if (jvm_signal_installed && sigused) {
    /* jvm has installed its signal handler for this signal. */
    /* Save the handler. Don't really install it. */
    if (oact != NULL) {
      *oact = sact[sig];                // [6a] 返回保存的处理器（欺骗调用者）
    }
    if (act != NULL) {
      sact[sig] = *act;                 // [6b] 保存新处理器到链中
    }

    signal_unlock();
    return 0;                           // [6c] 假装安装成功
  } 
  // ========== 分支 2: JVM 正在安装信号期间 ==========
  else if (jvm_signal_installing) {
    /* jvm is installing its signal handlers. Install the new
     * handlers and save the old ones. */
    res = call_os_sigaction(sig, act, &oldAct);  // [7a] 真正安装到内核
    sact[sig] = oldAct;                          // [7b] 保存旧处理器
    if (oact != NULL) {
      *oact = oldAct;                            // [7c] 返回旧处理器
    }

    sigaddset(&jvmsigs, sig);                    // [7d] 标记为 JVM 占用

    signal_unlock();
    return res;
  } 
  // ========== 分支 3: 与 JVM 无关的信号 ==========
  else {
    /* jvm has no relation with this signal (yet). Install the
     * the handler. */
    res = call_os_sigaction(sig, act, oact);     // [8] 直接透传给内核

    signal_unlock();
    return res;
  }
}
```

### 1.5 JVM 与 libjsig 的协作接口

```c
// jsig.c:304-328

/* The three functions for the jvm to call into. */
void JVM_begin_signal_setting() {
  signal_lock();
  sigemptyset(&jvmsigs);         // [1] 清空 JVM 信号集合
  jvm_signal_installing = true;  // [2] 设置"正在安装"标志
  tid = pthread_self();          // [3] 记录 JVM 线程 ID
  signal_unlock();
}

void JVM_end_signal_setting() {
  signal_lock();
  jvm_signal_installed = true;   // [4] 设置"已安装"标志
  jvm_signal_installing = false; // [5] 清除"正在安装"标志
  pthread_cond_broadcast(&cond); // [6] 唤醒所有等待的线程
  signal_unlock();
}

struct sigaction *JVM_get_signal_action(int sig) {
  allocate_sact();
  /* Does race condition make sense here? */
  if (sigismember(&jvmsigs, sig)) {
    return &sact[sig];           // [7] 返回信号链上下一个处理器
  }
  return NULL;
}
```

```cpp
void os::Linux::install_signal_handlers() {
    // 保证只会安装一次
    if (!signal_handlers_are_installed) {
        signal_handlers_are_installed = true;

        // ========== 1. 查找 libjsig 接口 ==========
        typedef void (*signal_setting_t)();
        signal_setting_t begin_signal_setting = NULL;
        signal_setting_t end_signal_setting = NULL;
        
        // 通过 dlsym 查找 libjsig.so 提供的三个函数
        begin_signal_setting = CAST_TO_FN_PTR(signal_setting_t,
                                              dlsym(RTLD_DEFAULT, "JVM_begin_signal_setting"));
        if (begin_signal_setting != NULL) {
            end_signal_setting = CAST_TO_FN_PTR(signal_setting_t,
                                                dlsym(RTLD_DEFAULT, "JVM_end_signal_setting"));
            get_signal_action = CAST_TO_FN_PTR(get_signal_t,
                                               dlsym(RTLD_DEFAULT, "JVM_get_signal_action"));
            libjsig_is_loaded = true;
        }
        
        // ========== 2. 通知 libjsig 开始安装 ==========
        if (libjsig_is_loaded) {
            (*begin_signal_setting)();  // JVM_begin_signal_setting()
        }
        
        // ========== 3. 安装各个信号处理器 ==========
        set_signal_handler(SIGSEGV, true);  // 段错误 → NPE/SOE
        set_signal_handler(SIGPIPE, true);  // 管道断裂 → 忽略
        set_signal_handler(SIGBUS, true);   // 总线错误 → SOE
        set_signal_handler(SIGILL, true);   // 非法指令 → 错误报告
        set_signal_handler(SIGFPE, true);   // 浮点异常 → ArithmeticException
        set_signal_handler(SIGXFSZ, true);  // 文件过大 → 忽略
        
        // ========== 4. 通知 libjsig 安装完成 ==========
        if (libjsig_is_loaded) {
            (*end_signal_setting)();  // JVM_end_signal_setting()
        }
    }
}
```

```cpp
void os::Linux::set_signal_handler(int sig, bool set_installed) {
    struct sigaction oldAct;
    
    // [1] 先查询当前处理器（不修改）
    sigaction(sig, NULL, &oldAct);
    
    // [2] 检查是否已被其他代码占用
    void* oldhand = get_signal_handler(&oldAct);
    
    if (oldhand != NULL && oldhand != CAST_FROM_FN_PTR(void*, SIG_DFL) &&
        oldhand != CAST_FROM_FN_PTR(void*, SIG_IGN)) {
        // 有人已经安装了处理器！
        if (AllowUserSignalHandlers) {
            // 允许用户处理器，只记录日志
            if (libjsig_is_loaded) {
                // libjsig 已处理信号链
            } else {
                // 警告：信号处理器可能被覆盖
            }
        } else {
            // 不允许，但已经安装了，记录警告
        }
    }
    
    // [3] 构建新的信号处理配置
    struct sigaction sigAct;
    sigfillset(&(sigAct.sa_mask));           // 阻塞所有信号（处理期间）
    sigAct.sa_sigaction = signalHandler;     // 设置处理函数
    sigAct.sa_flags = SA_SIGINFO | SA_RESTART;  // 使用 sa_sigaction，自动重启系统调用
    
    // [4] 安装到内核
    sigaction(sig, &sigAct, &oldAct);
    
    // [5] 保存原始处理器供后续对比检查
    void* oldhand2 = get_signal_handler(&oldAct);
    assert(oldhand2 == oldhand, "no concurrent signal handler installation");
}
```

### 2.3 信号处理入口：signalHandler

```cpp
// src/hotspot/os/linux/os_linux.cpp:5180-5185

static void signalHandler(int sig, siginfo_t *info, void *uc) {
    assert(info != NULL && uc != NULL, "it must be old kernel");
    int orig_errno = errno;  // 保存 errno
    
    JVM_handle_linux_signal(sig, info, uc, true);  // 调用 JVM 统一处理函数
    
    errno = orig_errno;      // 恢复 errno（信号处理器规范要求）
}
```

**为什么保存和恢复 errno？**

```
场景：应用程序正在执行系统调用
  
  应用程序代码: read(fd, buf, size)
                   │
                   ▼
  内核: 准备返回 EINTR（被信号中断）
         │
         ▼ 突然收到 SIGSEGV（如 NullPointerException）
  内核: 调用 signalHandler()
         │
         └──> 信号处理...如果修改了 errno
         │
  返回:  errno 不再是 EINTR，应用程序困惑

解决方案：信号处理器必须保持 errno 不变
```

```cpp
// 简化伪代码
int JVM_handle_linux_signal(int signo, siginfo_t* siginfo, void* ucontext, int abort_flag) {
    Thread* t = Thread::current_or_null();
    address addr = (address)siginfo->si_addr;  // 触发信号的地址
    
    switch(signo) {
        case SIGSEGV:
        case SIGBUS:
            // 检查 1: 是否是 NPE（访问地址 < 4096）
            if (addr < (address)os::vm_page_size()) {
                // 生成 NullPointerException
                // 修改 ucontext 中的 PC，使其指向异常处理代码
                return 1;  // 已处理
            }
            
            // 检查 2: 是否是栈溢出（访问栈保护页）
            if (t && t->is_in_stack_guard_pages(addr)) {
                // 生成 StackOverflowError
                return 1;
            }
            
            // 检查 3: 是否是 Safepoint Polling Page
            if (addr == (address)SafepointSynchronize::get_polling_page()) {
                // 进入安全点
                return 1;
            }
            
            // 无法识别，是严重错误
            if (abort_flag) {
                VMError::report_and_die(t, signo, addr, siginfo, ucontext);
            }
            return 0;  // 未处理，需要链式调用
            
        case SIGFPE:
            // 生成 ArithmeticException
            return 1;
            
        case SIGPIPE:
        case SIGXFSZ:
            // 忽略，让系统调用返回错误
            return 1;
    }
    return 0;
}
```

### 2.5 崩溃处理：crash_handler

```cpp
// src/hotspot/os/posix/vmError_posix.cpp:106-142

static void crash_handler(int sig, siginfo_t* info, void* ucVoid) {
    // [1] 解除信号阻塞（防止递归崩溃）
    sigset_t newset;
    sigemptyset(&newset);
    sigaddset(&newset, sig);
    for (int i = 0; i < NUM_SIGNALS; i++) {
        sigaddset(&newset, SIGNALS[i]);  // 解除所有同步信号
    }
    os::Posix::unblock_thread_signal_mask(&newset);
    
    // [2] 获取程序计数器
    ucontext_t* const uc = (ucontext_t*) ucVoid;
    address pc = (uc != NULL) ? os::Posix::ucontext_get_pc(uc) : NULL;
    
    // [3] 修正 PC（某些信号的 si_addr 是异常地址而非 PC）
    if (sig == SIGILL || sig == SIGFPE) {
        pc = (address) info->si_addr;
    }
    
    // [4] SafeFetch 故障处理（特殊用途）
    if (uc && pc && StubRoutines::is_safefetch_fault(pc)) {
        os::Posix::ucontext_set_pc(uc, 
            StubRoutines::continuation_for_safefetch_fault(pc));
        return;
    }
    
    // [5] 致命错误报告
    VMError::report_and_die(NULL, sig, pc, info, ucVoid);
}
```

```
│  │       ├─ sact[SIGSEGV] = native_handler  [保存到链中]                │   │
│  │       └─ return 0  [假装安装成功，实际内核仍用 JVM 处理器]            │   │

│  │           ├─ 不是栈溢出                                               │   │
```
