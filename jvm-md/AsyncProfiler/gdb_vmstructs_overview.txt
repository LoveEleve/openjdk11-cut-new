# 统计 gHotSpotVMStructs 条目数
b VM::init
commands
  set $count = 0
  set $base = (unsigned long)gHotSpotVMStructs
  set $s = (unsigned long)gHotSpotVMStructEntryArrayStride
  while (*(char**)($base + $count * $s) != 0)
    set $count = $count + 1
  end
  printf "\n===== gHotSpotVMStructs TOTAL ENTRIES = %d =====\n", $count

  set $count2 = 0
  set $base2 = (unsigned long)gHotSpotVMTypes
  set $s2 = (unsigned long)gHotSpotVMTypeEntryArrayStride
  while (*(char**)($base2 + $count2 * $s2) != 0)
    set $count2 = $count2 + 1
  end
  printf "gHotSpotVMTypes TOTAL ENTRIES = %d\n", $count2

  printf "\n===== SAMPLE: gHotSpotVMStructs[0..4] =====\n"
  set $i = 0
  while $i < 5
    set $e = $base + $i * $s
    printf "[%d] type=%-20s field=%-25s offset=%lu\n", $i, *(char**)($e + 0), *(char**)($e + 8), *(unsigned long*)($e + 32)
    set $i = $i + 1
  end
  printf "\n===== SAMPLE: gHotSpotVMStructs 中 JavaThread 相关 =====\n"
  set $i = 0
  while (*(char**)($base + $i * $s) != 0)
    set $e = $base + $i * $s
    set $t = *(char**)($e + 0)
    if ($t[0] == 'J' && $t[1] == 'a' && $t[2] == 'v' && $t[3] == 'a' && $t[4] == 'T')
      printf "[%d] type=%-20s field=%-25s offset=%lu\n", $i, $t, *(char**)($e + 8), *(unsigned long*)($e + 32)
    end
    set $i = $i + 1
  end
  continue
end
